#include <iostream>
using namespace std;

int main()
{
    // №2
    int n;
    long long fact = 1; // для хранения факториала при шагах
    long long F = 1;    // для итогового числа

    cout << "Введите n: ";
    cin >> n;
    for (int i = 1; i <= n; i++) 
    {
        fact = fact * i; // считаем факториал i 
        F = F * fact;    // умножаем на i! к общему произведению 
    }
    cout << "F(n) = " << F << endl;

    // №3
    int A, B;

    cout << "Введите A (0 или 1): ";
    cin >> A;

    B = 2 - A; // если A=0, получаем 2; если A=1, получаем 1
    cout << "B = " << B << endl;


    // №4
    int a, b;

    cout << "Количество элементов A: ";
    cin >> a;

    cout << "Количество элементов B: ";
    cin >> b;

    int A[100], B[100];

    cout << "Введите элементы массива A по возрастанию:" << endl;
    for (int i = 0; i < a; i++)
    {
        cin >> A[i];
    }

    cout << "Введите элементы массива B по возрастанию:" << endl;
    for (int i = 0; i < b; i++)
    {
        cin >> B[i]; 
    }

    int i = 0; // текущий элемент массива A
    int j = 0; // текущий элемент массива B

    cout << "Объединенный массив: ";

    for (int k = 0; k < a + b; k++)
    {
        // если массив A закончился, берем элементы из B
        if (i >= a)
        {
            cout << B[j] << " ";
            j++;
        }
        // если массив B закончился, берем элементы из A
        else if (j >= b)
        {
            cout << A[i] << " ";
            i++;
        }
        // сравниваем текущие элементы двух массивов
        else if (A[i] < B[j])
        {
            cout << A[i] << " ";
            i++;
        }
        else
        {
            cout << B[j] << " ";
            j++;
        }
    }

    cout << endl;


    return 0;
}
